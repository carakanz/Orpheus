using System;
using System.IO;
using Orpheus.Core.Geometry;
using Orpheus.Core.Sim;
using Orpheus.Core.World;
using Orpheus.Net.Session;
using Orpheus.Save;
using Xunit;

namespace Orpheus.Tests
{
    /// <summary>
    /// Разовый генератор файла формата 11 для теста C-011 (запускается на коде слияния task/E2-15, 4848c8c — два слоя, до
    /// многоклеточной мебели): плоский мир 256², зерно 1; игрок по сети ставит предметами из руки деревянный пол 3 × 2 тайла,
    /// каменную стену в западные половины x = 130 поверх пола, сундук со сдвигом (+1, 0) в тайле 126,130, верстак 1 × 1 в
    /// 129,132, кровать 1 × 1 в 124,132, факел со сдвигом (+1, +1) в 132,130. Запуск: копия кода (git archive) + этот файл как .cs,
    /// <c>GOLD_OUT=папка dotnet test --filter Make_Format11_Layers_File</c>.
    /// </summary>
    public class MakeV11Layers
    {
        [Fact]
        public void Make_Format11_Layers_File()
        {
            WorldModel world = FlatWorldGenerator.Instance.Generate(MapShape.World(256), 1);
            byte[] baseBytes = WorldSnapshot.WriteBase(world);
            using var rig = new EditTests.EditRig(world, _ => EditTests.TileCenter(128, 128));
            NetClient c = rig.Join();
            rig.Run(5);
            var inv = rig.Sim.GetEntity(c.EntityId).Inv;
            inv.SetCell(2, Items.WoodFloor, 40);
            inv.SetCell(3, Items.StoneWall, 6);
            inv.SetCell(4, Items.Chest, 1);
            inv.SetCell(5, Items.Workbench, 1);
            inv.SetCell(6, Items.Bed, 1);
            inv.SetCell(7, Items.Torch, 1);
            Assert.True(rig.Sim.SetInventory(c.EntityId, inv));
            rig.Run(3);

            WorldPos self = rig.Sim.GetEntity(c.EntityId).Mover.Position;
            c.Place(new TileCoord(130, 126), new TileCoord(132, 126), BuildDefs.WoodFloorKey, self, 2);
            rig.Run(4);
            c.Place(new TileCoord(130, 127), new TileCoord(132, 127), BuildDefs.WoodFloorKey, self, 2);
            rig.Run(4);
            c.Place(new TileCoord(130, 126), new TileCoord(130, 127), ObjectId.StoneWall, self, 3, 0 | (2 << 2)); // западные половины, на полу
            rig.Run(4);
            c.Place(new TileCoord(126, 130), new TileCoord(126, 130), ObjectId.Chest, self, 4, 1); // СЗ ячейка — восточная половина 126
            rig.Run(4);
            c.Place(new TileCoord(129, 132), new TileCoord(129, 132), ObjectId.Workbench, self, 5, 0);
            rig.Run(4);
            c.Place(new TileCoord(124, 132), new TileCoord(124, 132), ObjectId.Bed, self, 6, 0);
            rig.Run(4);
            c.Place(new TileCoord(132, 130), new TileCoord(132, 130), ObjectId.Torch, self, 7, 3); // сдвиг (+1, +1)
            rig.Run(400);

            var chars = new[]
            {
                new CharacterRecord { IsHost = true, Name = "Орфей", EntityId = c.EntityId, Inv = rig.Sim.GetEntity(c.EntityId).Inv },
            };

            byte[] save = WorldSave.Build(WorldSave.Capture(rig.Sim, WorldSnapshot.StoredHash(baseBytes), "формат11-слои", chars), 5, 0);
            Assert.Equal(11u, SaveContainer.Read(save).Header.FormatVersion);
            string dir = Environment.GetEnvironmentVariable("GOLD_OUT")!;
            File.WriteAllBytes(Path.Combine(dir, "format11_layers.save"), save);
            File.WriteAllBytes(Path.Combine(dir, "format11_layers.base"), baseBytes);
            File.WriteAllText(Path.Combine(dir, "format11_layers.hash.txt"), StateHash.Compute(rig.Sim).Total.ToString("x16"));
        }
    }
}
