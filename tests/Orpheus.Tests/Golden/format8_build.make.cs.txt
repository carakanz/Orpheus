using System;
using System.IO;
using Orpheus.Core.Geometry;
using Orpheus.Core.Movement;
using Orpheus.Core.Sim;
using Orpheus.Core.World;
using Orpheus.Net.Session;
using Orpheus.Save;
using Orpheus.WorldGen;
using Xunit;

namespace Orpheus.Tests
{
    /// <summary>
    /// Разовый генератор файла формата 8 с постройками для теста C-005 (запускается на коде слияния task/E2-3, 3846d7a —
    /// до E2-5, ключ пола в заявке = id пола): плоский мир 256², зерно 1, полоса воды; игрок ставит молотом по сети
    /// (тот же путь, что ЛКМ) три деревянных пола, три каменных, три настила на воде, три деревянные и три каменные стены;
    /// по одной стене каждого вида повреждено наполовину.
    /// </summary>
    public class MakeV8Build
    {
        private static readonly InputFrame HammerHold = new InputFrame(Input8.None, 0, 0, 2);

        [Fact]
        public void Make_Format8_Build_File()
        {
            WorldModel world = FlatWorldGenerator.Instance.Generate(MapShape.World(256), 1);
            for (int x = 120; x <= 136; x++)
            {
                world.SetBaseTile(new TileCoord(x, 125), FloorId.Water, ObjectId.Empty);
            }

            byte[] baseBytes = WorldSnapshot.WriteBase(world);
            using var rig = new EditTests.EditRig(world, _ => EditTests.TileCenter(128, 128));
            NetClient c = rig.Join();
            rig.Run(5);
            var inv = rig.Sim.GetEntity(c.EntityId).Inv;
            inv.Add(Material.Wood, 40);
            inv.Add(Material.Stone, 40);
            Assert.True(rig.Sim.SetInventory(c.EntityId, inv));
            rig.InputOf = _ => HammerHold;
            rig.Run(3);

            WorldPos self = rig.Sim.GetEntity(c.EntityId).Mover.Position;
            c.Place(new TileCoord(130, 128), new TileCoord(132, 128), FloorId.WoodFloor, self);
            rig.Run(4);
            c.Place(new TileCoord(130, 130), new TileCoord(132, 130), FloorId.StoneFloor, self);
            rig.Run(4);
            c.Place(new TileCoord(130, 125), new TileCoord(132, 125), FloorId.Platform, self);
            rig.Run(4);
            c.Place(new TileCoord(125, 127), new TileCoord(125, 129), ObjectId.WoodWall, self);
            rig.Run(4);
            c.Place(new TileCoord(123, 127), new TileCoord(123, 129), ObjectId.StoneWall, self);
            rig.Run(4);
            for (int i = 0; i < 3; i++)
            {
                Assert.Equal(FloorId.WoodFloor, rig.Sim.World.GetFloor(new TileCoord(130 + i, 128)));
                Assert.Equal(FloorId.StoneFloor, rig.Sim.World.GetFloor(new TileCoord(130 + i, 130)));
                Assert.Equal(FloorId.Platform, rig.Sim.World.GetFloor(new TileCoord(130 + i, 125)));
                Assert.Equal(ObjectId.WoodWall, ObjectId.IdOf(rig.Sim.World.GetObject(new TileCoord(125, 127 + i))));
                Assert.Equal(ObjectId.StoneWall, ObjectId.IdOf(rig.Sim.World.GetObject(new TileCoord(123, 127 + i))));
            }

            Assert.Equal(40 - 3 - 3 - 6, rig.Sim.GetEntity(c.EntityId).Inv.CountAt(Material.Wood));
            Assert.Equal(40 - 3 - 6, rig.Sim.GetEntity(c.EntityId).Inv.CountAt(Material.Stone));
            Assert.True(rig.Sim.DamageBuilding(new TileCoord(125, 129), 60)); // 120 → 60
            Assert.True(rig.Sim.DamageBuilding(new TileCoord(123, 129), 150)); // 300 → 150
            rig.Run(400); // флаг «под ударом» снят — в файле только прочность

            var chars = new[]
            {
                new CharacterRecord { IsHost = true, Name = "Орфей", EntityId = c.EntityId, Inv = rig.Sim.GetEntity(c.EntityId).Inv },
            };

            byte[] save = WorldSave.Build(WorldSave.Capture(rig.Sim, WorldSnapshot.StoredHash(baseBytes), "формат8-постройки", chars), 4, 0);
            Assert.Equal(8u, SaveContainer.Read(save).Header.FormatVersion);
            string dir = Environment.GetEnvironmentVariable("V8_OUT")!;
            File.WriteAllBytes(Path.Combine(dir, "format8_build.save"), save);
            File.WriteAllBytes(Path.Combine(dir, "format8_build.base"), baseBytes);
            File.WriteAllText(Path.Combine(dir, "format8_build.hash.txt"), StateHash.Compute(rig.Sim).Total.ToString("x16"));
        }
    }
}
